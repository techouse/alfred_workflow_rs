4/// Error type used by all fallible crate APIs.
5#[derive(Debug, thiserror::Error)]
6pub enum Error {
7    /// Automatic cache seconds were outside Alfred's accepted range.
8    #[error("automatic cache seconds must be between {min} and {max}, got {seconds}")]
9    InvalidAutomaticCacheSeconds {
10        /// Provided seconds value.
11        seconds: u64,
12        /// Inclusive minimum accepted seconds.
13        min: u64,
14        /// Inclusive maximum accepted seconds.
15        max: u64,
16    },
17
18    /// Typed action had no `text`, `url`, `file`, or `auto` value.
19    #[error("typed action must include at least one of text, url, file, or auto")]
20    EmptyTypedAction,
21
22    /// An item tried to contain both `arg` and `action`.
23    #[error("item cannot include both arg and action")]
24    ArgAndAction,
25
26    /// A modifier key set was empty.
27    #[error("modifier key set cannot be empty")]
28    EmptyModifierKeySet,
29
30    /// A modifier key was not one of Alfred's supported keys.
31    #[error("unknown modifier key `{0}`")]
32    UnknownModifierKey(String),
33
34    /// File cache max entries was zero.
35    #[error("cache max entries must be greater than zero, got {entries}")]
36    InvalidMaxCacheEntries {
37        /// Provided max-entry count.
38        entries: usize,
39    },
40
41    /// File cache backend or metadata access failed.
42    #[error("failed to access cache: {0}")]
43    Cache(String),
44
45    /// Plist user configuration parsing failed.
46    #[error("failed to parse user configuration plist: {0}")]
47    UserConfiguration(String),
48
49    /// GitHub updater repository URL was not a valid `github.com/owner/repo` URL.
50    #[error(
51        "GitHub repository URL must be hosted on github.com and include owner/repo, got `{url}`"
52    )]
53    InvalidGithubRepositoryUrl {
54        /// Provided repository URL.
55        url: String,
56    },
57
58    /// GitHub updater current version was not a strict semantic version.
59    #[error("current version must be a strict semantic version, got `{version}`")]
60    InvalidCurrentVersion {
61        /// Provided current version.
62        version: String,
63    },
64
65    /// Version parsing failed.
66    #[error("failed to parse version")]
67    Version(#[from] semver::Error),
68
69    /// URL parsing failed.
70    #[error("failed to parse URL")]
71    Url(#[from] url::ParseError),
72
73    /// HTTP request or response body handling failed.
74    #[error("HTTP request failed: {0}")]
75    Http(String),
76
77    /// JSON rendering or parsing failed.
78    #[error("failed to render JSON")]
79    Json(#[from] serde_json::Error),
80
81    /// Filesystem, process, or output writer operation failed.
82    #[error("filesystem, process, or output writer operation failed: {0}")]
83    Io(#[from] std::io::Error),
84}
