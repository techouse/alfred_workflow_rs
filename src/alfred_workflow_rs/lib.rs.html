1//! Strict Rust port of the Dart `alfred_workflow` package.
2//!
3//! The crate is library-first and focuses on Alfred Script Filter JSON,
4//! workflow output, file caching, plist user defaults, and GitHub updater
5//! behavior with Dart-compatible wire formats.
6//!
7//! ```
8//! use alfred_workflow_rs::{Item, Result, Workflow};
9//!
10//! # fn main() -> Result<()> {
11//! let mut workflow = Workflow::new();
12//! workflow.add_item(Item::with_arg("Search Google", "https://www.google.com"))?;
13//!
14//! assert_eq!(
15//!     workflow.to_json_string()?,
16//!     r#"{"items":[{"title":"Search Google","type":"default","valid":false,"arg":"https://www.google.com"}]}"#
17//! );
18//! # Ok(())
19//! # }
20//! ```
